name: Refresh datos.json

on:
  schedule:
    # 11:00 UTC = 08:00 ART. GitHub puede demorar las tareas programadas cuando
    # tiene carga, asi que no esperes que corra exactamente a esa hora.
    - cron: "0 11 * * *"
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: refresh-datos
  cancel-in-progress: false

jobs:
  refresh:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v5

      - uses: actions/setup-python@v6
        with:
          python-version: "3.12"
          cache: pip

      - name: Instalar dependencias
        run: pip install -r requirements.txt

      - name: Generar datos.json
        env:
          GCP_SA_KEY: ${{ secrets.GCP_SA_KEY }}
        run: python scripts/build_data.py --out datos.json

      # El control corre ANTES del commit, con el datos.json viejo todavia en
      # git, para poder comparar. Si encuentra algo que requiere decision, el
      # workflow falla y GitHub manda mail automaticamente.
      - name: Controlar los datos
        id: control
        run: python scripts/check_data.py

      # Se commitea aunque el control haya fallado: los datos nuevos son validos
      # igual, lo que falla es que hay algo para mirar.
      - name: Commitear si cambio
        if: always() && steps.control.outcome != 'cancelled'
        run: |
          if git diff --quiet -- datos.json; then
            echo "Sin cambios en datos.json."
            exit 0
          fi
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add datos.json
          git commit -m "Refresh datos.json ($(date -u +%Y-%m-%d))"
          git push
